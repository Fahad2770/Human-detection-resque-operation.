# 1. Fetch Fixed Channel Status (Vitals & RSSI)
@st.cache_data(ttl=2)
def get_thingspeak_status():
    try:
        response = requests.get(fixed_url)
        data = response.json()
        feeds = data['feeds']
        if feeds:
            latest = feeds[-1]
            val = float(latest['field1'])
            created_at = latest['created_at']
            return val, created_at
    except:
        pass
    return -70.0, None
